; Вывод:
;   -1          бесконечно много корней
;    0          корней нет
;    1  x       один корень
;    2  x1 x2   два корня
;
; Память: [0] a  [1] b  [2] c  [3] D  [4] 2a  [5] x1  [6] x2  [9] sqrt(D)

; ---------- коэффициенты: x^2 - 5x + 6 = 0 -> 2, 3, 2 ----------
PUSH 1
POP [0]             ; a = 1
PUSH 0
PUSH 5
SUB
POP [1]             ; b = -5
PUSH 6
POP [2]             ; c = 6

; ---------- a == 0 ? ----------
PUSH [0]
PUSH 0
JNE a_nonzero

PUSH [1]
PUSH 0
JNE linear          ; a == 0, b != 0

PUSH [2]
PUSH 0
JNE no_roots        ; a == 0, b == 0, c != 0
PUSH 0
PUSH 1
SUB
OUT                 ; -1: бесконечно много корней
HLT

no_roots:
PUSH 0
OUT                 ; корней нет
HLT

linear:             ; x = -c / b
PUSH 0
PUSH [2]
SUB
PUSH [1]
DIV
PUSH 1
OUT                 ; сначала количество корней
OUT
HLT

; ---------- a != 0: D = b*b - 4*a*c ----------
a_nonzero:
PUSH [1]
PUSH [1]
MUL
PUSH 4
PUSH [0]
MUL
PUSH [2]
MUL
SUB
POP [3]

PUSH [3]
PUSH 0
JB no_roots         ; D < 0

PUSH [3]
PUSH 0
JNE two_roots       ; D > 0

PUSH 0              ; D == 0: x = -b / (2a)
PUSH [1]
SUB
PUSH 2
PUSH [0]
MUL
DIV
PUSH 1
OUT
OUT
HLT

two_roots:          ; D > 0: x = (-b +- sqrt(D)) / (2a)
PUSH [3]
SQRT
POP [9]
PUSH 2
PUSH [0]
MUL
POP [4]

PUSH 0
PUSH [1]
SUB
PUSH [9]
ADD
PUSH [4]
DIV
POP [5]             ; x1

PUSH 0
PUSH [1]
SUB
PUSH [9]
SUB
PUSH [4]
DIV
POP [6]             ; x2

PUSH 2
OUT
PUSH [5]
OUT
PUSH [6]
OUT
HLT

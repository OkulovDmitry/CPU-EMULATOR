; Вывод:
;   -1          бесконечно много корней
;    0          корней нет
;    1  x       один корень
;    2  x1 x2   два корня
;
; Память: [0] a  [1] b  [2] c  [3] D  [4] 2a  [5] x1  [6] x2
;         [9] sqrt(D)  [10] аргумент f_zero  [11] результат f_zero

; Несколько тестов и ожидаемый вывод:
;
;
;
 
; =====================================================
; Основной цикл: ввод коэффициентов, решение, вывод, запрос остановить программу или принять новые коэффициенты
; =====================================================
main_loop:
IN
POP [0]             ; a
IN
POP [1]             ; b
IN
POP [2]             ; c
 
CALL f_solve        ; печатает ответ
 
IN                  ; спрашиваем: 1 - ещё раз, иначе выход
PUSH 1
JNE finish          ; введено не 1 -> конец
PUSH 0
PUSH 1
JNE main_loop       ; JMP main_loop
 
finish:
HLT
 
 
; =====================================================
; f_solve: решает уравнение с коэффициентами из [0], [1], [2]
; и печатает ответ
; =====================================================
f_solve:
; ---------- a = 0 ? ----------
PUSH [0]
POP [10]
CALL f_zero
PUSH [11]
PUSH 0
JNE a_zero
 
; ---------- a != 0: D = b*b - 4*a*c ----------
PUSH [1]
PUSH [1]
MUL
PUSH 4
PUSH [0]
MUL
PUSH [2]
MUL
SUB
POP [3]
 
PUSH [3]
POP [10]
CALL f_zero
PUSH [11]
PUSH 0
JNE one_root        ; D = 0
 
PUSH [3]
PUSH 0
JB no_roots         ; D < 0
 
; ---------- D > 0: x = (-b +- sqrt(D)) / (2a) ----------
PUSH [3]
SQRT
POP [9]             ; sqrt(D)
PUSH 2
PUSH [0]
MUL
POP [4]             ; 2a
 
PUSH 0
PUSH [1]
SUB
PUSH [9]
ADD
PUSH [4]
DIV
POP [5]             ; x1 = (-b + sqrt(D)) / 2a
 
PUSH 0
PUSH [1]
SUB
PUSH [9]
SUB
PUSH [4]
DIV
POP [6]             ; x2 = (-b - sqrt(D)) / 2a
 
PUSH 2
OUT
PUSH [5]
OUT
PUSH [6]
OUT
RET
 
one_root:           ; D = 0: x = -b / (2a)
PUSH 0
PUSH [1]
SUB
PUSH 2
PUSH [0]
MUL
DIV
PUSH 1
OUT                 ; сначала количество корней
OUT                 ; потом корень (он лежал в стеке)
RET
 
; ---------- a = 0: линейное уравнение ----------
a_zero:
PUSH [1]
POP [10]
CALL f_zero
PUSH [11]
PUSH 0
JNE b_zero          ; b = 0
 
PUSH 0              ; x = -c / b
PUSH [2]
SUB
PUSH [1]
DIV
PUSH 1
OUT
OUT
RET
 
b_zero:             ; a = 0 и b = 0: решение зависит от c
PUSH [2]
POP [10]
CALL f_zero
PUSH [11]
PUSH 0
JNE inf_roots       ; c = 0
 
no_roots:
PUSH 0
OUT                 ; корней нет
RET
 
inf_roots:
PUSH 0
PUSH 1
SUB
OUT                 ; -1: корней бесконечно много
RET
 
 
; =====================================================
; f_zero: [11] = 1, если |[10]| < 1e-9, иначе 0
; =====================================================
f_zero:
PUSH [10]
PUSH 0.000000001
JB fz_below         ; x < eps -> проверим x > -eps
PUSH 0
POP [11]
RET
fz_below:
PUSH 0
PUSH 0.000000001
SUB                 ; -eps
PUSH [10]
JB fz_yes           ; -eps < x
PUSH 0
POP [11]
RET
fz_yes:
PUSH 1
POP [11]
RET

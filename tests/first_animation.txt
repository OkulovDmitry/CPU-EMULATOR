; бегущая точка по первой строке экрана
PUSH 0
POP AX          ; номер клетки с желтой точкой
move:
PUSH 14         ; цвет желтый
POP [AX]
PUSH 0
POP BX          ; счетчик кадров задержки
wait:
DRAW
PUSH BX
PUSH 1
ADD
POP BX
PUSH BX
PUSH 10         ; проверка что набралось достаточно кадров
JB wait         ; если BX < 10 прыгаем вверх на wait
PUSH 0
POP [AX]        ; стираем точку
PUSH AX         ; Увеличиваем индекс пикселя AX на 1:
PUSH 1
ADD
POP AX          ; AX = AX + 1 (перемещаемся на следующую клетку)
PUSH AX
PUSH 16         ; Предполагается, что ширина экрана VRAM_W = 16 пикселей
JB move         ; Если AX < 16, прыгаем в начало 'move' и рисуем точку в новой позиции
HLT
